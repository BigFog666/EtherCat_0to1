cmake_minimum_required(VERSION 3.28)
project(EtherCAT_Hardware_Project LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(PROJECT_FIRMWARE "Build STM32F407ZE firmware" OFF)
if(PROJECT_FIRMWARE)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/firmware/generated/GCCSources.cmake")
    message(FATAL_ERROR "Run tools/prepare_firmware.py first")
  endif()
  include(firmware/generated/GCCSources.cmake)
  add_executable(ethercat_joint ${BOARD_SOURCES} firmware/startup_f407.c firmware/syscalls.c)
  set_target_properties(ethercat_joint PROPERTIES SUFFIX ".elf")
  target_include_directories(ethercat_joint PRIVATE ${BOARD_INCLUDES})
  target_compile_definitions(ethercat_joint PRIVATE USE_STDPERIPH_DRIVER STM32F40_41xxx)
  target_compile_options(ethercat_joint PRIVATE -Og -g3 -ffunction-sections -fdata-sections -fno-common)
  target_link_options(ethercat_joint PRIVATE
    "-T${CMAKE_CURRENT_SOURCE_DIR}/firmware/stm32f407ze.ld"
    -nostartfiles --specs=nano.specs --specs=nosys.specs -Wl,--gc-sections
    "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/ethercat_joint.map")
  target_link_libraries(ethercat_joint PRIVATE m)
  add_custom_command(TARGET ethercat_joint POST_BUILD
    COMMAND ${CMAKE_OBJCOPY} -O ihex $<TARGET_FILE:ethercat_joint> ${CMAKE_CURRENT_BINARY_DIR}/ethercat_joint.hex
    COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:ethercat_joint> ${CMAKE_CURRENT_BINARY_DIR}/ethercat_joint.bin
    COMMAND ${CMAKE_SIZE} $<TARGET_FILE:ethercat_joint>)
else()
  add_library(joint_core STATIC common/joint.c common/pdo.c)
  target_include_directories(joint_core PUBLIC common)
  target_compile_options(joint_core PRIVATE -Wall -Wextra -Werror -Wpedantic)
  enable_testing()
  add_executable(test_joint tests/test_joint.c)
  target_link_libraries(test_joint PRIVATE joint_core)
  target_compile_options(test_joint PRIVATE -Wall -Wextra -Werror -Wpedantic)
  add_test(NAME joint_core COMMAND test_joint)
  add_executable(offline_demo master/offline_demo.c)
  target_link_libraries(offline_demo PRIVATE joint_core)
  add_library(demo_sequence STATIC master/demo_sequence.c)
  target_include_directories(demo_sequence PUBLIC master)
  target_link_libraries(demo_sequence PUBLIC joint_core)
  add_executable(test_sequence tests/test_sequence.c)
  target_link_libraries(test_sequence PRIVATE demo_sequence)
  add_test(NAME master_sequence_simulated_pdo COMMAND test_sequence)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SOEM/CMakeLists.txt")
    set(SOEM_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SOEM")
  else()
    set(SOEM_DEFAULT "${CMAKE_CURRENT_SOURCE_DIR}/../EtherCAT_Master_Lab/vendor/SOEM")
  endif()
  set(SOEM_SOURCE_DIR "${SOEM_DEFAULT}" CACHE PATH "Fixed SOEM source directory")
  if(NOT EXISTS "${SOEM_SOURCE_DIR}/CMakeLists.txt")
    message(FATAL_ERROR "Existing SOEM source missing; set SOEM_SOURCE_DIR")
  endif()
  set(SOEM_BUILD_SAMPLES OFF CACHE BOOL "" FORCE)
  add_subdirectory("${SOEM_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/soem")
  add_executable(joint_master master/joint_master.c)
  target_link_libraries(joint_master PRIVATE demo_sequence soem)
  target_compile_options(joint_master PRIVATE -Wall -Wextra -Werror -Wpedantic)
  add_executable(eepromtool "${SOEM_SOURCE_DIR}/samples/eepromtool/eepromtool.c")
  target_link_libraries(eepromtool PRIVATE soem)
endif()
